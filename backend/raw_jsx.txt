<script>
  // RPG State Management
  let currentXP = 1240;
  const maxXP = 1500;
  let crowns = 2840;
  let pendingCount = 3;

  function updateTreasury(addedCrowns) {
    crowns += addedCrowns;
    const formatted = crowns.toLocaleString();
    const tCrowns = document.getElementById('treasury-crowns');
    const hCrowns = document.getElementById('header-crowns');
    const pCrowns = document.getElementById('pill-crowns');
    if (tCrowns) tCrowns.textContent = formatted;
    if (hCrowns) hCrowns.textContent = formatted;
    if (pCrowns) pCrowns.textContent = formatted;
  }

  // Floating Particle Generator
  function spawnFloatingParticle(text, originEl, color = 'text-gold-bright') {
    const container = document.getElementById('particle-container');
    if (!container) return;
    const particle = document.createElement('div');
    particle.className = `absolute z-50 font-cinzel font-bold text-sm pointer-events-none xp-particle-floating ${color} drop-shadow-[0_2px_8px_rgba(0,0,0,0.9)]`;
    particle.textContent = text;
    
    // Position inside module
    particle.style.left = '50%';
    particle.style.top = '40%';
    container.appendChild(particle);
    setTimeout(() => particle.remove(), 1100);
  }

  // Surge Arcane Essence XP
  function surgeArcaneEssence(amount = 120, label = '+120 XP') {
    const xpCurrentEl = document.getElementById('xp-current');
    const xpPercentEl = document.getElementById('xp-percent');
    const xpBarFill = document.getElementById('xp-bar-fill');
    const xpRemaining = document.getElementById('xp-remaining-text');
    
    currentXP = Math.min(maxXP, currentXP + amount);
    const newPercent = ((currentXP / maxXP) * 100).toFixed(1);
    
    if (xpCurrentEl) xpCurrentEl.textContent = currentXP.toLocaleString();
    if (xpPercentEl) xpPercentEl.textContent = `${newPercent}%`;
    if (xpBarFill) xpBarFill.style.width = `${newPercent}%`;
    if (xpRemaining) {
      const diff = maxXP - currentXP;
      xpRemaining.textContent = diff > 0 ? `+${diff} XP to Level 08 (The Nightseer)` : 'Rite of Ascension Primed!';
    }
    
    // Visual celebration flash
    xpBarFill.classList.add('ring-2', 'ring-gold-bright', 'brightness-125');
    spawnFloatingParticle(label, xpBarFill);
    setTimeout(() => {
      xpBarFill.classList.remove('ring-2', 'ring-gold-bright', 'brightness-125');
    }, 800);
  }

  // Interactive Surge Button & XP Bar clicks
  const surgeBtn = document.getElementById('surge-xp-btn');
  if (surgeBtn) {
    surgeBtn.addEventListener('click', () => surgeArcaneEssence(150, '+150 XP SURGE!'));
  }
  const xpContainer = document.getElementById('xp-bar-container');
  if (xpContainer) {
    xpContainer.addEventListener('click', () => surgeArcaneEssence(80, '+80 XP'));
  }

  // Complete Quest with Reward Banner & Audio/Haptic visual pulse
  document.querySelectorAll('.quest-seal-btn').forEach((btn) => {
    btn.addEventListener('click', function(e) {
      e.stopPropagation();
      const article = this.closest('article');
      const title = article.querySelector('h3');
      const xpVal = parseInt(article.dataset.xp || '80', 10);
      const crownsVal = parseInt(article.dataset.crowns || '25', 10);
      
      if (!article.classList.contains('opacity-75')) {
        this.innerHTML = '<span className="material-symbols-outlined text-base text-gold-bright">verified</span> <span>RITE SEALED</span>';
        this.classList.remove('bg-gradient-to-r', 'from-crimson-deep', 'to-[#8b0e0e]', 'hover:brightness-125');
        this.classList.add('bg-[#150101]', 'text-sage-muted', 'border-[#D4AF37]/30');
        
        title.classList.add('line-through', 'text-sage-muted');
        article.classList.add('opacity-75', 'bg-[#1a0204]/60');

        // Dynamic State Changes
        surgeArcaneEssence(xpVal, `+${xpVal} XP`);
        updateTreasury(crownsVal);
        
        pendingCount = Math.max(0, pendingCount - 1);
        const pCountEl = document.getElementById('pending-quests-count');
        const qSubtitle = document.getElementById('quest-header-subtitle');
        if (pCountEl) pCountEl.textContent = pendingCount;
        if (qSubtitle) qSubtitle.textContent = `${pendingCount} Pending Rites • ${4 - pendingCount} Sealed`;

        // Prepend deed to chronicle
        const deedsList = document.getElementById('deeds-timeline');
        if (deedsList) {
          const newDeed = document.createElement('div');
          newDeed.className = 'relative flex items-start gap-3 transition-all duration-500 animate-fadeIn';
          newDeed.innerHTML = `
            <div className="w-5 h-5 rounded-full bg-crimson-deep text-gold-bright flex items-center justify-center text-xs flex-shrink-0 z-10 border border-[#D4AF37]/50 shadow-sm">
              <span className="material-symbols-outlined text-[12px]">check</span>
            </div>
            <div className="flex-1 space-y-0.5">
              <div className="flex justify-between items-baseline">
                <span className="font-cinzel text-sm font-semibold text-ivory-warm">Quest Sealed: ${title.textContent.trim()}</span>
                <span className="font-mono text-[11px] text-gold-bright">Just now</span>
              </div>
              <p className="font-sans text-xs text-sage-muted">Yielded +${xpVal} XP and +${crownsVal} Crowns.</p>
            </div>
          `;
          deedsList.insertBefore(newDeed, deedsList.children[1]);
        }
      }
    });
  });

  // Vault Influx Harvest Micro-Action
  const harvestBtn = document.getElementById('vault-influx-btn');
  if (harvestBtn) {
    harvestBtn.addEventListener('click', () => {
      updateTreasury(50);
      spawnFloatingParticle('+50 Crowns Harvested', harvestBtn, 'text-gold-amber');
      harvestBtn.classList.add('brightness-150', 'scale-95');
      setTimeout(() => harvestBtn.classList.remove('brightness-150', 'scale-95'), 200);
    });
  }

  // 3D Tilt effect on Attribute Cards via mousemove
  document.querySelectorAll('.stat-card-3d').forEach((card) => {
    card.addEventListener('mousemove', (e) => {
      const rect = card.getBoundingClientRect();
      const x = e.clientX - rect.left - rect.width / 2;
      const y = e.clientY - rect.top - rect.height / 2;
      const mult = 14;
      const xRot = -(y / (rect.height / 2)) * mult;
      const yRot = (x / (rect.width / 2)) * mult;
      card.style.transform = `perspective(1000px) rotateX(${xRot.toFixed(2)}deg) rotateY(${yRot.toFixed(2)}deg) scale3d(1.035, 1.035, 1.035)`;
    });
    card.addEventListener('mouseleave', () => {
      card.style.transform = 'perspective(1000px) rotateX(0deg) rotateY(0deg) scale3d(1, 1, 1)';
    });
  });

  // Audio Toggle Simulation
  const audioBtn = document.getElementById('ambient-audio-btn');
  const audioIcon = document.getElementById('audio-icon');
  const audioLabel = document.getElementById('audio-label');
  let audioPlaying = true;

  if (audioBtn) {
    audioBtn.addEventListener('click', () => {
      audioPlaying = !audioPlaying;
      if (audioPlaying) {
        audioIcon.textContent = 'volume_up';
        audioLabel.textContent = 'Sanctum Ambient';
        audioBtn.classList.remove('opacity-60');
      } else {
        audioIcon.textContent = 'volume_off';
        audioLabel.textContent = 'Muted';
        audioBtn.classList.add('opacity-60');
      }
    });
  }

  // Keyboard shortcut listener for 'N' (Forge Quest)
  window.addEventListener('keydown', (e) => {
    if (e.key === 'n' || e.key === 'N') {
      if (['INPUT', 'TEXTAREA'].includes(document.activeElement.tagName)) return;
      const forgeBtn = document.getElementById('forge-btn');
      if (forgeBtn) {
        forgeBtn.classList.add('scale-95', 'brightness-150');
        setTimeout(() => forgeBtn.classList.remove('scale-95', 'brightness-150'), 180);
      }
    }
  });
</script></main>
